# elkbench built-in start for the Analog Four MKII (OS 1.56 with analoghealth), emulator only.
# A gate batch runs it on a fresh emulator, keeps the end state as a snapshot, and runs hardware
# plans from there (docs/testing-practices.md, "Messy starts and contrast runs"). Names and slots
# follow the example write policy (examples/write-policy): `~T ` test names, test slots 121-128.
format = 1
unit = a4
emulator = yes
name = start dirty
out = runs
note = the card's own project active, pattern A02, T1 OSC1 LEVEL changed unsaved, LFO page open

[start]
# Boot prompts away (the third NO lands on the home screen and does nothing)
wait 3
tap NO
wait 0.5
tap NO
wait 0.5
tap NO
wait 1.0
shot start-home.png

[pattern A02]

wait 1.5
shot a02-home.png
