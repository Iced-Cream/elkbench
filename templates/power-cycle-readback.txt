# Template: a value kept across a power cycle, read back before and after it.
# The `power-cycle` step runs in the emulator too: elkbench switches analog-emu off with its
# power-off save (the OS writes its working state to the card), then cold-boots it on that card.
# On the unit the run waits at the step until the person has power cycled it by hand and said
# "done" in the thread; the go request names the step. A pass so means "kept across a power
# cycle", not "kept with nothing happening". After the step the unit is at its start-up screen:
# reach the page again from a known page.
format = 1
name = kept after the power cycle
unit = rytm
refs = refs/power-cycle-readback
out = runs
note = ~T SETUP active, with the value to keep set by an earlier plan

[start]
# The readback before the power cycle: the value is there now. Reach the page from a known page
# (TRIG, then LFO), and compare only the rows that show the kept value. A row that shows which
# page is open passes on a default project too, and proves nothing about what was kept.
wait 3
tap NO
wait 0.5
tap NO
wait 0.5
tap NO
wait 1.0
keys TRK(BD)
wait 0.4
tap TRIG
wait 0.8
tap LFO
wait 1.5
shot before.png
# the value the earlier plan set, e.g. a knob's dial (rows 8-23): stop otherwise. Leave out the
# label rows below: just after a turn, the value pop-up covers them, and would differ for a
# reason that has nothing to do with what was kept
# The contrast: on a fresh card, or from the built-in dirty, occupied and testslot-cleared starts,
# the earlier plan set no value, so this check has to stop the run. This plan's own pass comes from
# run_plan after that plan (the gate batch runs only its starts, and those must stop before the
# power cycle).
say contrast fresh, dirty, occupied, testslot-cleared: a default project has no such value, so the readback stops the run
expect-screen kept-value 1 rows=8-23

[kept after the power cycle]
say proves set-and-kept: the value reads back the same after a power cycle
power-cycle
# the start-up screen first, then the same way to the page
wait 3
tap NO
wait 0.5
tap NO
wait 0.5
tap NO
wait 1.0
keys TRK(BD)
wait 0.4
tap TRIG
wait 0.8
tap LFO
wait 1.5
shot after.png
expect-screen kept-value 1 rows=8-23
