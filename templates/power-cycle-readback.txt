# Template: the plan after a power cycle by the person's hand, proving what the power cycle had to keep.
# The run pauses for the power cycle and waits for the person's typed "done" in the thread; the go
# request names it ("hand power cycle, no emulator gate"), and every plan before and after has its
# emulator pass. This plan's emulator pass runs on an emulator that never lost power: its start check
# must be one that a default project fails.
format = 1
name = kept after the power cycle
unit = rytm
refs = refs/power-cycle-readback
out = runs
note = ~T SETUP active: read only after the power cycle

[start]
# The readback is the start check: reach the page from a known page (TRIG, then LFO), and compare
# only the rows that show the kept value. A row that shows which page is open passes on a default
# project too, and proves nothing about what was kept.
wait 3
tap NO
wait 0.5
tap NO
wait 0.5
tap NO
wait 1.0
keys TRK(BD)
wait 0.4
tap TRIG
wait 0.8
tap LFO
wait 1.5
shot readback.png
# the value the earlier plan set and saved, e.g. the LFO's DST field (rows 8-31): stop otherwise.
# The contrast: from a fresh card, or the built-in dirty and occupied starts, no value was kept, so
# the readback has to stop the run. This plan's own pass comes from run_plan after the plan that
# set the value (the gate batch runs only its starts).
say contrast fresh, dirty, occupied: a default project has no kept value, so the readback stops the run
expect-screen kept-value 1 rows=8-31

[the rest of the plan]
say proves set-and-kept: a second kept value, read back the same way
tap FX
wait 0.8
tap TRIG
wait 0.8
tap LFO
wait 1.5
expect-screen kept-value-2 1 rows=8-31
