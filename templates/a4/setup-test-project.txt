# Template: build the run's own test project on an Analog Four MKII (OS 1.56 + analoghealth), and
# prove it. Copy it beside your plans and change only what the comments name. Rules behind it:
# docs/testing-practices.md, "Plan checker and templates". The route is the Rytm's: MANAGE
# PROJECTS > INIT NEW into the cleared test slot, then LOAD FROM it.
#
# Before this plan, the run backs up and clears the test slots that hold a project (outside the
# plan for now), e.g. for 121:
#   data get /projects/121      (the backup, into the run's evidence folder)
#   data clear /projects/121
# List each in the go request as a write: "slot 121 (holds X at listing time: backed up and erased)".
# The start check sees 121-128, so all eight must be empty. The LEVEL turn in the last section
# makes the active project unsaved, and the load then discards it (answered NO to the save): list
# that in the go request too.
format = 1
name = setup test project in 121
unit = a4
refs = refs/a4-setup-test-project     # this plan's own references: never share them with another plan
out = runs
note = ~T SETUP (slot 121, INIT NEW) active, unsaved changes none

[start]
# Boot prompts away, then GLOBAL > PROJECT > MANAGE PROJECTS (third row). The list holds every
# slot, 001-128, eight rows on screen; it is entered from its end (page downs first) so a
# remembered cursor does not matter, and UP x7 puts 121 on the top row with 122-128 below it.
wait 3
tap NO
wait 0.5
tap NO
wait 0.5
tap NO
wait 1.0
keys GLOBAL,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,YES,UP,UP,UP,DOWN,DOWN,YES
wait 0.8
keys FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),UP,UP,UP,UP,UP,UP,UP
wait 0.6
shot start-manage-121.png
# The start check: 121-128 all empty, which only the run's own clears produced. Record it in the
# emulator after emulator-only steps have filled 121 and cleared it. The active project's row
# carries a small marker at the left of its name field, even when its slot has been cleared, and
# which row has it depends on the unit: so the check is in two parts that leave that column out,
# the slot numbers and the name fields. The scrollbar (cols 122-127) is left out too.
expect-screen manage-121-128-empty 2 rows=0-63 cols=0-28
expect-screen manage-121-128-empty 2 rows=0-63 cols=31-121

[setup: INIT NEW into 121]
say proves setup: a new project in 121, named by the plan
# RIGHT opens the slot panel (YES on the list only ticks the slot); an empty slot's panel starts
# with INIT NEW, which makes the project at once and asks no name. The panel then stays open as a
# full slot's panel, cursor on CLEAR: nothing may press YES after it. The current project stays
# active. The name is the plan's own, at most 15 characters; save-project writes it over the data
# store. save-project does not mark kit 1 `~T` here (the kit it would mark is the active one).
save-project 121 "~T SETUP" GLOBAL,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,YES,UP,UP,UP,DOWN,DOWN,YES,FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),UP,UP,UP,UP,UP,UP,UP,RIGHT,UP,UP,UP,UP,UP,UP,YES
wait 2
tap NO
wait 0.5
tap NO
wait 0.5
tap NO
wait 0.5
tap NO
wait 1.0
shot init-new.png

[setup: LOAD FROM 121, and prove the load by content]
say proves setup: the new project is the active one (its home screen, compared whole, not its name)
# A LEVEL turn on T1's OSC1 page first, so the active project has unsaved changes and both load
# questions come every time (with none, the OS loads without a word). Remote turns under about
# 3 ticks are dropped.
tap T1
wait 0.5
tap OSC1
wait 0.6
turn LEVEL 8 4
wait 0.5
keys GLOBAL,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,YES,UP,UP,UP,DOWN,DOWN,YES
wait 0.8
keys FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),UP,UP,UP,UP,UP,UP,UP
wait 0.6
# a full slot's panel: CLEAR, DELETE, RENAME, LOAD FROM, SAVE TO, TOGGLE
tap RIGHT
wait 0.6
keys UP,UP,UP,UP,UP,UP,DOWN,DOWN,DOWN
wait 0.4
shot load-from.png
tap YES
wait 0.8
shot save-prompt.png
# SAVE '<name>' BEFORE LOADING? Y/N. Always NO: YES saves the active project over its own slot,
# outside the plan's slots (the checker refuses anything else after a save-before-load check).
# The box is as wide as the project's name, so the check compares the prompt's fixed second line
# only, inside the narrowest box. When the active project came from 121 itself, the OS skips this
# question: then the check stops the run with nothing written but 121.
expect-screen save-before-loading 2 rows=34-40 cols=22-106
tap NO
wait 0.8
shot proceed.png
# ANY CHANGES WILL BE LOST. PROCEED WITH LOADING? Y/N; YES loads 121. The box's inside only:
# the list and panel show beside it.
expect-screen proceed-with-loading 2 rows=21-43 cols=8-120
tap YES
# the load ends back on the list with the panel open: four NOs reach home (panel, list, PROJECT,
# GLOBAL), and a spare NO on the home screen does nothing
wait 6
tap NO
wait 0.6
tap NO
wait 0.6
tap NO
wait 0.6
tap NO
wait 0.6
tap NO
wait 1.5
shot loaded.png
# The content proof: the home screen of a new project (K001:KIT 1, BPM:120.0), compared whole. A
# name check passes on a copy of any project.
expect-screen new-home 3
