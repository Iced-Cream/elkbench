# Template: a value kept across a power cycle on an Analog Four MKII, read back before and after it.
# The `power-cycle` step runs in the emulator too: elkbench switches analog-emu off with its
# power-off save (the OS writes its working state to the card), then cold-boots it on that card.
# On the unit the run waits at the step until the person has power cycled it by hand and said
# "done" in the thread; the go request names the step. A pass so means "kept across a power
# cycle", not "kept with nothing happening". After the step the unit is at its start-up screen:
# reach the page again from a known page.
format = 1
name = kept after the power cycle
unit = a4
refs = refs/a4-power-cycle-readback
out = runs
note = ~T SETUP active, with the value to keep set and saved by an earlier plan

[start]
# The readback before the power cycle: the value is there now. Reach the page from a known page (a
# track key, OSC1, then the page key), and compare only the rows that show the kept value. A row
# that shows which page is open passes on a default project too, and proves nothing about what
# was kept. ENV and LFO each cycle between two pages (ENV1/ENV2, LFO1/LFO2): entered from another
# page, the first press opens the first one.
wait 3
tap NO
wait 0.5
tap NO
wait 0.5
tap NO
wait 1.0
tap T1
wait 0.5
tap OSC1
wait 0.6
tap LFO
wait 1.0
shot before.png
# The value the earlier plan set and saved, e.g. LFO1's first DST (knob G; LFO 1 P 1/2 has SPD MUL
# FAD SPH MOD on top, WAV DST DEP DST DEP below), in the rows below the title bar: stop otherwise.
# A DST knob opens the destination list as a popup, and the value is set only by YES: the earlier
# plan confirms it with YES and checks it is shown before it saves.
# The contrast: on a fresh card the earlier plan set no value, so this check has to stop the run.
# This plan's own pass comes from run_plan after that plan.
say contrast fresh: a default project has no such value, so the readback stops the run
expect-screen kept-value 1 rows=8-63

[kept after the power cycle]
say proves set-and-kept: the value reads back the same after a power cycle
power-cycle
# the start-up screen first, then the same way to the page. The ping and the shot come before any
# key, so the run's evidence shows the unit answering and what it showed after the power cycle
# (on the unit the cycle itself rests on the person's "done").
wait 3
ping
shot after-power-cycle.png
tap NO
wait 0.5
tap NO
wait 0.5
tap NO
wait 1.0
tap T1
wait 0.5
tap OSC1
wait 0.6
tap LFO
wait 1.0
shot after.png
expect-screen kept-value 1 rows=8-63
