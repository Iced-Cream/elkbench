# Template: a value kept across a power cycle on a Digitone (OS 1.44 with a remote-key path, such as
# a digihealth build), read back before and after it. Copy it beside your plans and change only what
# the comments name.
# The `power-cycle` step runs in the emulator too: elkbench switches digiemu's supply off, the OS
# runs its own power-off save (its working state to the card), and digiemu boots the same OS again
# on that card. On the unit the run waits at the step until the person has power cycled it by hand
# and said "done" in the thread; the go request names the step. A pass so means "kept across a
# power cycle", not "kept with nothing happening".
#
# The run this plan belongs to starts from empty test slots: 121-128 hold nothing when the run
# begins (backed up and erased before it, or emptied at the last release). In it, setup-test-project
# builds ~T SETUP in 121, a plan of the run's own sets the value to keep (here: a turn of encoder A
# on track 1's TRIG page, with nothing saved), then this plan reads it back. The value is in the
# working state, not in a saved project: that is what a power cycle has to keep.
format = 1
name = kept after the power cycle
unit = digitone
refs = refs/digitone-power-cycle-readback   # this plan's own references: never share them with another plan
out = runs
note = ~T SETUP active, with the value to keep set by an earlier plan and not saved

[start]
# The readback before the power cycle: the value is there now. Reach the page from home (NO closes
# SETTINGS and its panels), then track 1, then TRIG, as the capture template does. The page title
# row (rows 0-9) shows "Trig Parameters" only for a moment, so compare rows 10-63: they show the
# page, the track and the encoders' values. Record the reference after the earlier plan's change.
tap NO
wait 0.4
tap NO
wait 0.4
tap NO
wait 0.4
tap NO
wait 1.0
tap T1
wait 0.4
tap TRIG
wait 1.5
shot before.png
# The contrast: on a fresh card (and, once the Digitone has them, from the built-in starts: the
# empty test slots one included) the earlier plan set no value, so this check has to stop the run.
# This plan's own pass comes from run_plan after that plan (the gate batch runs only its starts,
# and those must stop before the power cycle).
say contrast fresh: a default project has no such value, so the readback stops the run
expect-screen kept-value 1 rows=10-63

[kept after the power cycle]
say proves set-and-kept: the value reads back the same after a power cycle
power-cycle
# On the unit, an OS with remote keys answers them about 40 s after it starts; in the emulator
# elkbench waits for the OS before this part, and the wait costs nothing else.
wait 40
tap NO
wait 0.4
tap NO
wait 0.4
tap NO
wait 0.4
tap NO
wait 1.0
tap T1
wait 0.4
tap TRIG
wait 1.5
shot after.png
expect-screen kept-value 1 rows=10-63
