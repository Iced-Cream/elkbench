# Template: build the run's own test project on a Digitone (OS 1.44 with a remote-key path, such as a
# digihealth build), and prove it. Copy it beside your plans and change only what the comments name.
# Rules behind it: docs/testing-practices.md, "Plans: what a check must show". Paths found by trial
# in digiemu (2026-11-02).
#
# Before this plan, the run backs up and clears the test slot (outside the plan for now):
#   data get /projects/121      (the backup, into the run's evidence folder)
#   data clear /projects/121
# List both in the go request as writes: "slot 121 (holds X at listing time: backed up and erased)".
# The Digitone marks a test project by its pattern A01's name in a write policy, not a kit.
format = 1
name = setup test project in 121
unit = digitone
refs = refs/digitone-setup-test-project   # this plan's own references: never share them with another plan
out = runs
note = ~T SETUP (slot 121, INIT NEW) active, unsaved changes none

[start]
# Home first (NO closes SETTINGS and its panels), then MANAGE PROJECTS from the SETTINGS list's
# top (it does not wrap), then the cursor on 121 by a fixed count: FUNC(UP) and FUNC(DOWN) move
# eight slots, so 18 up reach the top whatever the remembered cursor, and 15 down put 121 in the
# list's bottom row.
tap NO
wait 0.4
tap NO
wait 0.4
tap NO
wait 0.4
tap NO
wait 1.0
keys SETTINGS,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,YES,UP,UP,UP,DOWN,DOWN,YES
wait 1.0
keys FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN)
wait 0.6
shot start-manage-121.png
# The start check: 121's name field in the bottom row (rows 56-63, cols 32-80) is the dashed line of
# an empty slot, which only the run's own clear produced. Record it in the emulator after an
# emulator-only step has filled and cleared 121, so the clearing path is tested too.
expect-screen manage-121-cleared 2 rows=56-63 cols=32-80

[setup: INIT NEW into 121]
say proves setup: a new project in 121, named by the plan
# RIGHT opens the slot's panel; on an empty slot it lists INIT NEW, SAVE TO, TOGGLE, and INIT NEW
# writes at once (no naming screen, no confirm). save-project names the slot over the data store
# afterwards. Never press YES in the list without the panel: it ticks the slot.
save-project 121 "~T SETUP" SETTINGS,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,YES,UP,UP,UP,DOWN,DOWN,YES,FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),RIGHT,UP,UP,UP,YES
wait 6
tap NO
wait 0.4
tap NO
wait 0.4
tap NO
wait 0.4
tap NO
wait 1.5
shot init-new.png

[setup: LOAD FROM 121, and prove the load by content]
say proves setup: the new project is the active one (its title row, not its name)
# The Digitone asks before loading only when the working project has unsaved changes, and then
# only once: ANY CHANGES WILL BE LOST. PROCEED WITH LOADING? Y/N (no SAVE ... BEFORE LOADING? was
# seen). A plan cannot branch on whether it comes, so this turn makes the working project modified,
# and the prompt always comes. It loses nothing the load would not lose anyway: YES there discards
# every unsaved change, so the go request names that.
turn A 40
wait 1
keys SETTINGS,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,YES,UP,UP,UP,DOWN,DOWN,YES
wait 1.0
keys FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN)
wait 0.6
tap RIGHT
wait 0.8
# on a filled slot the panel lists CLEAR, DELETE, RENAME, LOAD FROM, SAVE TO, TOGGLE
keys UP,UP,UP,UP,UP,UP,DOWN,DOWN,DOWN
wait 0.4
tap YES
wait 1
shot proceed.png
# the box's fixed text (rows 25-29 and 35-39) inside its frame
expect-screen proceed-with-loading 2 rows=20-44 cols=7-121
tap YES
# the load takes a while, and digirun gets no answer meanwhile
wait 8
tap NO
wait 0.6
tap NO
wait 0.6
tap NO
wait 0.6
tap NO
wait 1.5
shot loaded.png
# The content proof: the home title row (rows 0-9) of a new project reads A01 UNTITLED and 120.0
# (the factory project reads A01 IMAGINE and 86.0). A name check passes on a copy of any project.
expect-screen loaded-new-untitled 2 rows=0-9
