# Template: build the run's own test project on a Digitone (OS 1.44 with a remote-key path, such as a
# digihealth build), and prove it. Copy it beside your plans and change only what the comments name.
# Rules behind it: docs/testing-practices.md, "Plans: what a check must show". Paths found by trial
# in digiemu (2026-11-02).
#
# Not INIT NEW: after a data-store clear of the active project's slot (the slot lists empty, the
# active marker stays on it), INIT NEW into that slot writes a project that still carries records of
# the old one (one of its sounds, by name, in the bytes; found in digiemu, 2026-11-02). So, as on
# the A4, the new project comes from CREATE NEW (the LOAD PROJECT list's last row), SAVE PROJECT AS
# puts it into 121 and makes the saved copy active, and loading 121 proves it is active; its home
# title row proves it is new.
#
# The test slot must be empty when this plan starts. With `clear_at_release` covering it in
# config.toml (as in the example setup: 121-128), every release leaves it so, and the go request
# says so. Without it, the run backs up and clears the test slot before this plan (hand steps):
#   data get /projects/121      (the backup, into the run's evidence folder)
#   data clear /projects/121
# List both in the go request as writes: "slot 121, whatever it holds: backed up and erased". The
# plan passes from either start a unit may be in: 121 cleared while still the active project (the
# active marker stays on its row), or another project active and 121 empty. The A01 turns in the
# second and fourth sections make the working project unsaved, and CREATE NEW (answered NO to the
# save) and the load discard them: list that in the go request too.
# The Digitone marks a test project by its pattern A01's name in a write policy, not a kit.
format = 1
name = setup test project in 121
unit = digitone
refs = refs/digitone-setup-test-project   # this plan's own references: never share them with another plan
out = runs
# Every Digitone hardware plan starts from empty test slots: its own gate run starts from the
# built-in start testslot-cleared, the state every release leaves with clear_at_release (the last
# test project still the working project, its slot 121 cleared, the active marker on 121).
start = testslot-cleared
note = ~T SETUP (slot 121, CREATE NEW) active, unsaved changes none

[start]
# Home first (NO closes SETTINGS and its panels), then MANAGE PROJECTS from the SETTINGS list's
# top (it does not wrap), then the cursor on 121 by a fixed count: FUNC(UP) and FUNC(DOWN) move
# eight slots, so 18 up reach the top whatever the remembered cursor, and 15 down put 121 in the
# list's bottom row.
tap NO
wait 0.4
tap NO
wait 0.4
tap NO
wait 0.4
tap NO
wait 1.0
keys SETTINGS,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,YES,UP,UP,UP,DOWN,DOWN,YES
wait 1.0
keys FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN)
wait 0.6
shot start-manage-121.png
# The start check: 121's row (the list's bottom row, rows 56-63) shows an empty slot. The active
# project's row carries a small marker left of its name field (rows 58-60, cols 29-30), even when
# its slot has been cleared. digirun compares whole rows (it takes rows=, no cols=), so the check
# leaves the marker's rows out: rows 56-57 and 61-63 hold the slot number and the top and bottom
# rows of a name's letters, which an empty slot's dashed line does not light. Record it in the
# emulator, then see it fail with 121 filled.
# The contrast: from the built-in occupied and testslot starts, 121 holds a project, and the
# empty-slot check stops the run. From testslot-cleared (this plan's own start) and dirty, 121 is
# empty and the plan passes.
say contrast occupied, testslot: slot 121 holds a project, so the empty-slot check stops the run
expect-screen manage-121-empty 2 rows=56-57
expect-screen manage-121-empty 2 rows=61-63
# back home: the list, PROJECT, SETTINGS (a spare NO on the home screen does nothing)
tap NO
wait 0.4
tap NO
wait 0.4
tap NO
wait 0.4
tap NO
wait 1.0

[setup: CREATE NEW]
say proves setup: a new project is the working project (the load questions it answers)
# A turn first, so the working project has unsaved changes and both load questions come every
# time (with none, the OS loads without a word). Both come from either start, the save question
# naming the working project. They are answered without a screen check: digirun compares whole
# rows, and across these boxes' rows the save box is as wide as the project's name and the LOAD
# PROJECT list behind them shows as many rows as the unit holds projects. The answers are safe
# whatever comes: NO to the save, and a NO to PROCEED would only cancel, which the home check
# below then stops on.
turn A 40
wait 1
# a shot of the change (the turn names the parameter in the title row)
shot turned.png
# SETTINGS > PROJECT (top row) > LOAD PROJECT (top row) > CREATE NEW (the list's last row)
keys SETTINGS,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,YES,UP,UP,UP,YES
wait 1.0
keys FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN)
wait 0.6
shot createnew.png
tap YES
wait 1
shot save-before.png
# SAVE '<name>' BEFORE LOADING? Y/N. Always NO: YES saves the working project outside the plan's
# slots.
tap NO
wait 1
shot proceed.png
# ANY CHANGES WILL BE LOST. PROCEED WITH LOADING? Y/N; YES makes the new project.
tap YES
# CREATE NEW takes a moment, and digirun gets no answer meanwhile
wait 4
tap NO
wait 0.6
tap NO
wait 0.6
tap NO
wait 0.6
tap NO
wait 1.5
shot new-home.png
# A new project's home title row (rows 0-9) reads A01 UNTITLED and 120.0 (the factory project reads
# A01 IMAGINE and 86.0).
expect-screen new-untitled 2 rows=0-9

[setup: SAVE PROJECT AS 121]
say proves setup: the new project saved into 121 (proven active in the next section)
# SETTINGS > PROJECT > SAVE PROJECT AS (second row) > the slot list from 001: 18 FUNC(UP) reach the
# top, 15 FUNC(DOWN) put the cursor on 121 > YES (the naming screen) > YES (save). save-project
# names the slot over the data store afterwards; it leaves pattern A01's name as it is (the unit
# cannot confirm that the active project is the saved copy), so the home title row still reads
# UNTITLED.
save-project 121 "~T SETUP" SETTINGS,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,YES,UP,UP,UP,DOWN,YES,FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),YES,YES
wait 2
tap NO
wait 0.4
tap NO
wait 0.4
tap NO
wait 0.4
tap NO

shot saved.png

[setup: 121 is the active project]
say proves setup: loading 121 asks only PROCEED, which the OS does only when 121 is the active project
# The proof is the OS's own answer, not a name. With unsaved changes, LOAD FROM asks SAVE ...
# BEFORE LOADING? first when the chosen project is not the active one, and only PROCEED WITH
# LOADING? when it is. So: a turn, MANAGE PROJECTS, 121's LOAD FROM, YES, and the PROCEED box must
# be the first thing on screen. YES then reloads 121 itself.
turn A 40
wait 1
shot turned-again.png
keys SETTINGS,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,UP,YES,UP,UP,UP,DOWN,DOWN,YES
wait 1.0
keys FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(UP),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN),FUNC(DOWN)
wait 0.6
# 121's panel (RIGHT): CLEAR, DELETE, RENAME, LOAD FROM, SAVE TO, TOGGLE; LOAD FROM is the fourth
tap RIGHT
wait 0.8

wait 0.4
shot load-121.png
tap YES
wait 1
shot load-121-asks.png
# The box's rows (20-44) across the whole width: the box itself (cols 7-121, the same for every
# project) and, beside it, the ends of the slot numbers 116-120 and the panel's edge, which are
# fixed in MANAGE PROJECTS with 121 in the bottom row.
expect-screen manage-proceed-with-loading 2 rows=20-44
tap YES

wait 8
tap NO
wait 0.6
